const rect=(x:number,y:number,w:number,h:number):Selection=>({inverted:false,antialiased:false,shapes:[{kind:'rectangle',bounds:[x,y,w,h],feather:0,operation:'replace'}]})
function field(){const w=64,h=48,destination=new Uint8Array(w*h*4),donor=new Float32Array(w*h*4),coverage=new Uint8Array(w*h);for(let y=0;y<h;y++)for(let x=0;x<w;x++){const p=y*w+x;destination.set([145,145,145,255],p*4);const selected=x>=24&&x<40&&y>=16&&y<32,detail=selected?(x+y)%2?12:-12:0;donor.set([60+detail,60+detail,60+detail,255],p*4);if(selected)coverage[p]=255}return {w,h,destination,donor,coverage}}
function editorFixture(){const png=blankPNG(80,56);for(let y=0;y<56;y++)for(let x=0;x<80;x++)png.data.set([80+x,100+y,130+(x+y)%9,255],(y*80+x)*4);for(let y=20;y<30;y++)for(let x=45;x<55;x++)png.data.set([220,30,20,255],(y*80+x)*4);const editor=new Editor();editor.importPNG(encodeRaster(png),'Diffusion fixture');editor.apply({...editor.identity(),action:{type:'selection.set',selection:rect(45,20,10,10)}});const snapshot=editor.snapshot(),stage={...snapshot,selection:editor.state().selection??null},id=snapshot.manifest.layers[0].id;return {editor,stage,id,png}}
const patch=(id:string,extra:Partial<PatchAction>={}):PatchAction=>({type:'layer.patch',layerId:id,mode:'source',offset:[-30,0],adaptation:1,...extra})

 assert.equal(sampledHealingHalo(),2);assert.equal(sampledHealingHalo(1),2);assert.ok(sampledHealingHalo(7)>sampledHealingHalo(5))

test('Spatial Diffusion changes nonuniform exterior tone independently of adaptation strength',()=>{
 const {w,h,destination,donor,coverage}=field();for(let y=0;y<h;y++)for(let x=0;x<w;x++){const p=y*w+x;if(!coverage[p])destination.set([Math.abs(x-32)<10?80:190,Math.abs(y-24)<10?90:180,130,255],p*4)}
 const one=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:1}),five=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:5}),seven=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:7});assert.notDeepEqual(five,one);assert.notDeepEqual(seven,five)
 const raw=blendSampledHealing(destination,donor,coverage,w,h,{adaptation:0,diffusion:5});for(let p=0;p<coverage.length;p++)if(coverage[p])for(let c=0;c<3;c++)assert.equal(raw[p*4+c],donor[p*4+c])
})

test('Diffusion preserves alpha-locked hidden RGB, low-alpha exterior confidence and single opacity application',()=>{
 const {w,h,destination,donor,coverage}=field(),p=23*w+31;destination.set([17,39,61,0],p*4);for(let i=0;i<coverage.length;i++)if(!coverage[i]&&i%5===0)destination[i*4+3]=1
 const before=destination.slice(),full=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:5}),half=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:5,opacity:.5});assert.deepEqual(full.subarray(p*4,p*4+4),before.subarray(p*4,p*4+4));for(let i=0;i<coverage.length;i++){assert.equal(half[i*4+3],before[i*4+3]);for(let c=0;c<3;c++)assert.ok(Math.abs(half[i*4+c]-(before[i*4+c]+full[i*4+c])/2)<=1)}assert.deepEqual(destination,before)

test('Near-transparent exterior RGB has proportionate tone confidence and cannot poison the repair field',()=>{
 const {w,h,destination,donor,coverage}=field(),baseline=blendSampledHealing(destination,donor,coverage,w,h,{diffusion:5}),weak=destination.slice(),opaque=destination.slice()
 for(let y=12;y<16;y++)for(let x=24;x<40;x++){weak.set([250,0,200,1],(y*w+x)*4);opaque.set([250,0,200,255],(y*w+x)*4)}
 const low=blendSampledHealing(weak,donor,coverage,w,h,{diffusion:5}),strong=blendSampledHealing(opaque,donor,coverage,w,h,{diffusion:5});let weakMaximum=0,strongMaximum=0
 for(let p=0;p<coverage.length;p++)if(coverage[p])for(let c=0;c<3;c++){weakMaximum=Math.max(weakMaximum,Math.abs(low[p*4+c]-baseline[p*4+c]));strongMaximum=Math.max(strongMaximum,Math.abs(strong[p*4+c]-baseline[p*4+c]))}
 assert.ok(weakMaximum<=1);assert.ok(strongMaximum>5)
})
